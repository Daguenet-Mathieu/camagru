mkdir -p Services/volumes/volumemedia

bash Services/Server/scripts/ssl.sh

docker build Services/Server/ -t serv
docker build Services/service_test/ -t php_test
docker build Services/Media/ -t media
docker build Services/Mail/ -t mail

docker network create camagru

docker run -d --name app_php --network camagru php_test
docker run -d --name app_mail --network camagru mail

docker run -d --name app_media \
  --network camagru \
  -v ./Services/volumes/volumemedia:/app/volume \
  media

docker run -d -p 8080:80 -p 4433:443 \
  --name nginx \
  --network camagru \
  -v ./Services/volumes/volumemedia:/app/volume \
  serv
